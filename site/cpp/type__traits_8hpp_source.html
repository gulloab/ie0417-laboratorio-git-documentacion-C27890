200// Explicitly define char traits for unsigned char since it is not standard
201template<>
202struct char_traits<unsigned char> : std::char_traits<char>
203{
204 using char_type = unsigned char;
205 using int_type = uint64_t;
206
207 // Redefine to_int_type function
208 static int_type to_int_type(char_type c) noexcept
209 {
210 return static_cast<int_type>(c);
211 }
212
213 static char_type to_char_type(int_type i) noexcept
214 {
215 return static_cast<char_type>(i);
216 }
217
218 static constexpr int_type eof() noexcept
219 {
220 return static_cast<int_type>(std::char_traits<char>::eof());
221 }
222};
223
224// Explicitly define char traits for signed char since it is not standard
225template<>
226struct char_traits<signed char> : std::char_traits<char>
227{
228 using char_type = signed char;
229 using int_type = uint64_t;

231 // Redefine to_int_type function

233 {
234 // cast via unsigned char: sign-extending a negative char_type would make
235 // byte 0xFF indistinguishable from eof()
236 return static_cast<int_type>(static_cast<unsigned char>(c));
237 }
238

241 return static_cast<char_type>(i);
242 }
243
244 static constexpr int_type eof() noexcept

343template<typename T>
344struct is_range
345{
346 private:
347 using t_ref = typename std::add_lvalue_reference<T>::type;
348
349 using iterator = detected_t<result_of_begin, t_ref>;
350 using sentinel = detected_t<result_of_end, t_ref>;
351
352 // to be 100% correct, it should use https://en.cppreference.com/w/cpp/iterator/input_or_output_iterator
353 // and https://en.cppreference.com/w/cpp/iterator/sentinel_for
354 // but reimplementing these would be too much work, as a lot of other concepts are used underneath
355 static constexpr auto is_iterator_begin =
356 is_iterator_traits<iterator_traits<iterator>>::value;
357
358 public:
359 static constexpr bool value = !std::is_same<iterator, nonesuch>::value && !std::is_same<sentinel, nonesuch>::value && is_iterator_begin;
360};

483// SafeToCheck guards against types that trigger circular constraints when
484// std::ranges::view<T> is evaluated on GCC 12 / libstdc++ 12:
485// - iteration_proxy / iteration_proxy_value directly
486// - views wrapping the above (e.g. owning_view<iteration_proxy<...>>)
487// - views wrapping basic_json (e.g. ref_view<json>) — same circularity
488// via json's constructors → is_compatible_array_type → here

490// before any std::ranges concept is touched, so we use it for the checks.

492 !is_iteration_proxy_type<T>::value &&
493 !is_iteration_proxy_type<detected_t<range_value_t, T>>::value &&

495 !is_range_view_optional_type<T>::value >
